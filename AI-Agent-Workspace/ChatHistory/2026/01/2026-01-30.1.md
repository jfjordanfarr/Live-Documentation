## Concrete Change

```typescript
// In packages/shared/src/languages/syntax.ts

/**
 * Creates a synchronous comment-only stripper from a LanguageSyntax.
 * 
 * Use this for regex-based matchers (heuristics, oracles) that need to
 * preserve string content. For tree-sitter contexts where the AST handles
 * interpolated strings correctly, use syntax.stripCommentsAndStrings().
 */
export function createCommentsOnlyStripper(
  syntax: LanguageSyntax
): (content: string) => string {
  // Build regex from syntax.comments.line and syntax.comments.block
  return (content: string) => {
    let result = content;
    // Strip line comments
    for (const lineDelim of syntax.comments.line) {
      const escaped = lineDelim.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      result = result.replace(new RegExp(`${escaped}.*$`, 'gm'), '');
    }
    // Strip block comments
    for (const [open, close] of syntax.comments.block) {
      const escapedOpen = open.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      const escapedClose = close.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      result = result.replace(new RegExp(`${escapedOpen}[\\s\\S]*?${escapedClose}`, 'g'), '');
    }
    return result;
  };
}
```
