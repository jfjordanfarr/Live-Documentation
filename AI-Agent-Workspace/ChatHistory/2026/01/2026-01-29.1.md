## Proposed Common Interface: `LanguageSyntax`

```typescript
/**
 * Language-agnostic syntax helpers that any adapter/heuristic can use.
 * This is the "contract" that prevents us from re-discovering solutions.
 */
export interface LanguageSyntax {
  /** Unique language identifier */
  readonly id: string;
  
  /** File extensions this applies to */
  readonly extensions: readonly string[];
  
  /** Strip comments and string literals, returning code-only content */
  stripNonCode(content: string): string;
  
  /** Known stdlib/built-in identifiers to ignore */
  readonly stdlibSymbols: ReadonlySet<string>;
  
  /** Common variable names that cause noise */
  readonly commonVariables: ReadonlySet<string>;
  
  /** Check if a symbol reference exists in code (not comments/strings) */
  isSymbolReferenced(content: string, symbolName: string): boolean;
  
  /** Reserved keywords that aren't valid symbols */
  readonly keywords: ReadonlySet<string>;
}
```

### Implementation per Language

```typescript
// packages/shared/src/language/go.ts
export const GO_SYNTAX: LanguageSyntax = {
  id: "go",
  extensions: [".go"],
  stripNonCode: stripGoCommentsAndStrings,
  stdlibSymbols: GO_STDLIB_PACKAGES,
  commonVariables: GO_COMMON_VARIABLE_NAMES,
  isSymbolReferenced: (content, sym) => {
    const code = stripGoCommentsAndStrings(content);
    return new RegExp(`(?<![.])\\b${escapeRegExp(sym)}\\b`).test(code);
  },
  keywords: new Set(["func", "type", "var", "const", "package", "import", ...]),
};
```

### Then Both Systems Consume It

```typescript
// In Go heuristic
import { GO_SYNTAX } from "../language/go";

if (GO_SYNTAX.isSymbolReferenced(source.content, symbolName)) {
  emit({ target, ... });
}

// In Go adapter  
import { GO_SYNTAX } from "../language/go";

const codeOnly = GO_SYNTAX.stripNonCode(content);
```
